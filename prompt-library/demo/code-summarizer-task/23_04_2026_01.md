import java.time.Duration;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.function.Function;

public class AsyncCacheExecutor<K, V> {

    private final Map<K, CompletableFuture<V>> cache = new ConcurrentHashMap<>();
    private final ScheduledExecutorService scheduler = Executors.newScheduledThreadPool(1);
    private final ExecutorService executor = Executors.newFixedThreadPool(Runtime.getRuntime().availableProcessors());
    private final int maxRetries;

    public AsyncCacheExecutor(int maxRetries) {
        this.maxRetries = maxRetries;
    }

    public CompletableFuture<V> getOrCompute(
            K key,
            Function<K, V> task,
            Duration timeout
    ) {
        return cache.computeIfAbsent(key, k -> executeWithRetry(k, task, timeout));
    }

    private CompletableFuture<V> executeWithRetry(
            K key,
            Function<K, V> task,
            Duration timeout
    ) {
        AtomicInteger attempt = new AtomicInteger(0);
        CompletableFuture<V> result = new CompletableFuture<>();

        Runnable execution = new Runnable() {
            @Override
            public void run() {
                CompletableFuture
                        .supplyAsync(() -> task.apply(key), executor)
                        .orTimeout(timeout.toMillis(), TimeUnit.MILLISECONDS)
                        .whenComplete((value, error) -> {
                            if (error == null) {
                                result.complete(value);
                            } else if (attempt.incrementAndGet() <= maxRetries) {
                                scheduler.schedule(this, 100L * attempt.get(),
                                        TimeUnit.MILLISECONDS);
                            } else {
                                cache.remove(key);
                                result.completeExceptionally(error);
                            }
                        });
            }
        };

        execution.run();
        return result.thenApply(Objects::requireNonNull);
    }

    public Optional<V> getIfCompleted(K key) {
        return Optional.ofNullable(cache.get(key))
                .filter(CompletableFuture::isDone)
                .filter(f -> !f.isCompletedExceptionally())
                .map(CompletableFuture::join);
    }

    public void shutdown() {
        executor.shutdown();
        scheduler.shutdown();
        cache.clear();
    }
}
